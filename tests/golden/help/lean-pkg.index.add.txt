usage: lean-pkg index add [-h] [--from-registry] [--repo-dir REPO_DIR] [--url URL] [--tag TAG]
                          [--lower REL] [--comparator REL] [--license SPDX]
                          [--headline 'NAME :: FILE :: SUMMARY'] [--field-tag TAG]
                          [--formalization-yaml REL] [--consent-recorded] --by BY
                          [--allow-unpushed] [--notes NOTES] [--built-on-file FILE] [--replace]
                          [--index INDEX] [--json] [--commit] [--push] [--no-commit]
                          name

positional arguments:
  name

options:
  -h, --help            show this help message and exit
  --from-registry       start from the registry row (repo, receipts, license, palomar, doi,
                        reservoir)
  --repo-dir REPO_DIR   local clone to read from (never modified)
  --url URL             public https URL (a GitHub ssh URL is converted) (old spelling --repo,
                        kept working)
  --tag TAG             release tag at the pin or an evidence-only descendant (or a docs-only one
                        when lean-pkg-binding.yaml at the pin opts in)
  --lower REL           lower-tier receipt, .lean-receipts/...json
  --comparator REL      comparator receipt, .lean-receipts/...json
  --license SPDX
  --headline 'NAME :: FILE :: SUMMARY'
  --field-tag TAG       free field tag (repeatable)
  --formalization-yaml REL
  --consent-recorded    every coauthor agreed to the listing (the pointer stays private)
  --by BY               public handle of the person adding the entry
  --allow-unpushed      do not verify that the pin and evidence are on the public repository
                        (provisional entry)
  --notes NOTES
  --built-on-file FILE  YAML list of credits for third-party Lean code the package builds on
                        (name, url, authors; optional license, pin, relation, note); [] for none
  --replace             replace an existing entry of this name (a re-pin); its built_on is carried
                        over unless --built-on-file is given
  --index INDEX         index root (default: $LEAN_PKG_INDEX, else index/ beside the registry when
                        present, else ./index)
  --json                machine-readable JSON on stdout
  --commit              write commands: pull --ff-only, then commit the registry files (default:
                        $LEAN_PKG_COMMIT=1)
  --push                --commit, then push to origin (default: $LEAN_PKG_PUSH=1)
  --no-commit           never commit or push (overrides the env)
