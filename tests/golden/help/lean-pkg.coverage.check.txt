usage: lean-pkg coverage check [-h] [--file PATH] [--allow-private-manuscript] [--json] [--commit]
                               [--push] [--no-commit]
                               name

positional arguments:
  name

options:
  -h, --help            show this help message and exit
  --file PATH           check this working-copy file instead of coverage.yaml committed at HEAD
                        (claim ids and receipts are still read at HEAD)
  --allow-private-manuscript
                        accept manuscript.version 'unposted' (a private draft); the output says so
  --json                machine-readable JSON on stdout
  --commit              write commands: pull --ff-only, then commit the registry files (default:
                        $LEAN_PKG_COMMIT=1)
  --push                --commit, then push to origin (default: $LEAN_PKG_PUSH=1)
  --no-commit           never commit or push (overrides the env)
