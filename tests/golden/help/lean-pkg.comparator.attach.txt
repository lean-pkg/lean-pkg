usage: lean-pkg comparator attach [-h] [--receipt RECEIPT] [--refuse-private] [--refuse-dirty]
                                  [--replace] [--agent AGENT] [--json] [--commit] [--push]
                                  [--no-commit]
                                  name

positional arguments:
  name

options:
  -h, --help         show this help message and exit
  --receipt RECEIPT  path to comparator receipt JSON (required)
  --refuse-private   refuse (exit 1, nothing copied) when the receipt holds private strings;
                     default is a WARN line and the receipt is kept
  --refuse-dirty     refuse when tracked files are modified in the checkout; default is a WARN
                     line
  --replace          when .lean-receipts/ already holds a different comparator receipt for the
                     same commit: keep it as <c12>-comparator.replaced-<UTC stamp>.json, attach
                     the new one, record comparator-replaced
  --agent AGENT      the agent acting (recorded as the event's actor; another agent's live lease
                     then refuses)
  --json             machine-readable JSON on stdout
  --commit           write commands: pull --ff-only, then commit the registry files (default:
                     $LEAN_PKG_COMMIT=1)
  --push             --commit, then push to origin (default: $LEAN_PKG_PUSH=1)
  --no-commit        never commit or push (overrides the env)
