usage: lean-pkg claims check [-h] [--file PATH] [--json] [name]

positional arguments:
  name

options:
  -h, --help   show this help message and exit
  --file PATH  schema-check this claims file (e.g. claims.yaml before you commit it) instead of
               the registry cache; schema only (no checkout, so declarations are not looked up:
               `claims sync` and `gate` do that); reads no registry; NAME, if given, must be its
               package
  --json       machine-readable JSON on stdout
