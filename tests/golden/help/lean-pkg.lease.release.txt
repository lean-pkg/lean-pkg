usage: lean-pkg lease release [-h] [--agent AGENT] [--force] [--reason REASON] [--json] [--commit]
                              [--push] [--no-commit]
                              name

positional arguments:
  name

options:
  -h, --help       show this help message and exit
  --agent AGENT    the holder; with --force, the acting agent recorded as released_by
  --force          release another agent's lease; requires --reason
  --reason REASON  free-text reason stored with the event
  --json           machine-readable JSON on stdout
  --commit         write commands: pull --ff-only, then commit the registry files (default:
                   $LEAN_PKG_COMMIT=1)
  --push           --commit, then push to origin (default: $LEAN_PKG_PUSH=1)
  --no-commit      never commit or push (overrides the env)
