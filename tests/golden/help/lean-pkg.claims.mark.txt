usage: lean-pkg claims mark [-h] --from-comparator CONFIG.json
                            --status {reusable,internal,experimental} [--write] [--dry-run]
                            [--agent AGENT] [--json]
                            name

positional arguments:
  name

options:
  -h, --help            show this help message and exit
  --from-comparator CONFIG.json
                        the comparator configuration whose theorem_names select the claims
                        (relative paths are read from the checkout)
  --status {reusable,internal,experimental}
                        the status to set
  --write               edit the claims file in the checkout (only the status values; never
                        committed)
  --dry-run             print the changes; write nothing (the default)
  --agent AGENT         with --write: the lease holder, when the package is leased
  --json                machine-readable JSON on stdout
