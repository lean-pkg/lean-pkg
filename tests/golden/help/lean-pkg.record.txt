usage: lean-pkg record [-h] [--field KEY=VALUE] [--reason REASON] [--doi-track {palomar,doi-only}]
                       [--agent AGENT] [--json] [--commit] [--push] [--no-commit]
                       name event

positional arguments:
  name
  event                 one of: consent-recorded, palomar-preflight-pass, palomar-submitted,
                        palomar-review, palomar-registered, palomar-withdrawn, doi-minted,
                        reservoir-listed, mathlib-pr-opened, mathlib-pr-merged, or retract
                        (--field event=E --field ts=T --reason TEXT: mark an evidence entry
                        retracted)

options:
  -h, --help            show this help message and exit
  --field KEY=VALUE     event field (repeatable); required keys per event are listed by `lean-pkg
                        llms`
  --reason REASON       free-text reason stored with the event
  --doi-track, --track {palomar,doi-only}
                        doi-minted only: the DOI track for this event (default: the record's
                        track, else palomar); doi-only skips rule doi-in-palomar for a public
                        package
  --agent AGENT         the agent acting (recorded as the event's actor); another agent's live
                        lease then refuses
  --json                machine-readable JSON on stdout
  --commit              write commands: pull --ff-only, then commit the registry files (default:
                        $LEAN_PKG_COMMIT=1)
  --push                --commit, then push to origin (default: $LEAN_PKG_PUSH=1)
  --no-commit           never commit or push (overrides the env)
